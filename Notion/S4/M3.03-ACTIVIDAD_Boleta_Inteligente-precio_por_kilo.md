# 🧪 ACTIVIDAD: Boleta Inteligente (precio por kilo)

## 🎯 Objetivo

Que entiendan:

- Cómo transformar unidades (kg → precio)
- Uso de `double`
- Aplicación real de operaciones

---

# 💻 CÓDIGO

```java
import java.util.Scanner;

public class BoletaPorPeso {

    public static void main(String[] args) {

        Scanner scanner = new Scanner(System.in);

        System.out.println("=== BOLETA INTELIGENTE ===");

        // ========================
        // 🛒 PRODUCTO POR PESO
        // ========================
        System.out.print("\nIngrese nombre del producto: ");
        String producto = scanner.nextLine();

        System.out.print("Precio por kilo ($): ");
        double precioKilo = scanner.nextDouble();

        System.out.print("Cantidad en kilos (ej: 1.3): ");
        double kilos = scanner.nextDouble();

        // ========================
        // 🧮 CÁLCULO
        // ========================
        double totalProducto = precioKilo * kilos;

        // ========================
        // 🛒 PRODUCTO FIJO
        // ========================
        System.out.print("\nIngrese nombre de otro producto: ");
        scanner.nextLine(); // limpiar buffer
        String producto2 = scanner.nextLine();

        System.out.print("Precio unitario ($): ");
        double precio2 = scanner.nextDouble();

        System.out.print("Cantidad: ");
        int cantidad2 = scanner.nextInt();

        double total2 = precio2 * cantidad2;

        // ========================
        // 🧾 TOTALES
        // ========================
        double subtotal = totalProducto + total2;
        double iva = subtotal * 0.19;
        double totalFinal = subtotal + iva;

        // ========================
        // 🧾 BOLETA
        // ========================
        System.out.println("\n=== BOLETA ===");

        System.out.println(producto + " (" + kilos + " kg) = $" + totalProducto);
        System.out.println(producto2 + " x" + cantidad2 + " = $" + total2);

        System.out.println("\nSubtotal: $" + subtotal);
        System.out.println("IVA (19%): $" + iva);
        System.out.println("TOTAL: $" + totalFinal);

        System.out.println("\n=== GRACIAS POR SU COMPRA ===");

        scanner.close();
    }
}
```

    import java.util.Scanner;

            double precioUnitario = scanner.nextDouble();
    
            System.out.print("Ingrese cantidad de unidades: ");
            int cantidadUnitaria = scanner.nextInt();
    
            double totalProductoUnitario = precioUnitario * cantidadUnitaria;
    
            // TOTALES
            double subtotal = totalProductoKilo + totalProductoUnitario;
            double iva = subtotal * 0.19;

            // ========================

            System.out.print("\nIngrese producto unitario: ");

            System.out.print("Precio unitario: $");

            double totalUnitario = precioUnitario * cantidad;
